'use strict';
const E = require('./engine.js');

// Change one construction parameter while preserving the complete puzzle and its witness.
// Derived losing-choice counts can change with the deadline; these are effects, not extra edits.
function clockSpare(design, spare, options = {}) {
  if (!Number.isInteger(spare) || spare < 0) return { fail: 'invalidSpare' };
  if (!design.clock) return { fail: 'noClock' };
  const copy = JSON.parse(JSON.stringify(design));
  const way = E.clockShortest(E.game(copy), { nodes: options.nodes ?? 400000, ms: options.ms ?? 5000 });
  if (!way || way.unsure) return { fail: 'minimumUnproven' };
  const before = copy.clock.n;
  copy.clock.n = way.k + spare;
  const witness = E.replay(copy, copy.solution);
  if (!witness.ok) return { fail: 'witnessWouldChange', detail: witness };
  return { design: copy, descriptor: 'frame.spareGame', before: before - way.k, after: spare,
    proof: { minimum: way.k, witness: true, changed: ['clock.n'] } };
}
// Reframe an existing matching puzzle instead of rebuilding its pictures around
// a new clock route. Only the clock and the solver's witness may change. This is
// a construction control; the new board still needs static/trace/bot checks.
function clockTile(design, tile, spare, options = {}) {
  if (!design.clock) return { fail: 'noClock' };
  if (!Number.isInteger(tile) || tile < 0 || tile >= design.tiles.length) return { fail: 'invalidClockTile' };
  if (!Number.isInteger(spare) || spare < 1 || spare > 3) return { fail: 'invalidSpare' };
  if ((design.down || []).includes(tile)) return { fail: 'clockWouldBeHidden' };
  const copy = JSON.parse(JSON.stringify(design));
  copy.clock = { i: tile, n: 99 };
  const limits = { nodes: 400000, ms: Infinity, finishNodes: 300000, finishMs: Infinity, ...options };
  const gameWay = E.clockShortest(E.game(copy), limits);
  if (!gameWay || gameWay.unsure) return { fail: 'minimumUnproven' };
  if (gameWay.k + spare > 19) return { fail: 'clockTooDeep', minimum: gameWay.k };
  const strict = E.clockShortest(E.game(copy, { strict: true }), limits);
  if (!strict || strict.unsure) return { fail: 'strictMinimumUnproven' };
  if (strict.k !== gameWay.k) return { fail: 'peekShortcut', gameMinimum: gameWay.k, strictMinimum: strict.k };
  copy.clock.n = gameWay.k + spare;
  copy.solution = E.flatten(strict.path);
  if (!E.replay(copy, copy.solution).ok || !E.replay(copy, copy.solution, { strict: true }).ok) return { fail: 'witnessUnproven' };
  return { design: copy, descriptor: 'frame.clockTile', before: design.clock.i, after: tile,
    proof: { minimum: gameWay.k, strictMinimum: strict.k, spare, witness: true, changed: ['clock', 'solution'],
      preserved: ['tiles', 'faces', 'down', 'gold', 'obst'], scope: 'legal construction, not difficulty or quality acceptance' } };
}
// Add an explicit key prelude to a plain clock without redealing its puzzle.
function clockKeys(design, keys, spare, options = {}) {
  if(!design.clock)return {fail:'noClock'};
  if(!Array.isArray(keys)||keys.length!==2||keys[0]===keys[1]||keys.some(i=>!Number.isInteger(i)||i<0||i>=design.tiles.length||i===design.clock.i))return {fail:'invalidKeys'};
  if(![1,2,3].includes(spare))return {fail:'invalidSpare'};
  const ob=design.obst||{};
  const lockTile=options.lockTile??design.clock.i;
  if(!Number.isInteger(lockTile)||lockTile<0||lockTile>=design.tiles.length||keys.includes(lockTile))return {fail:'invalidLockTile'};
  if(ob.locks?.length||ob.keys?.length||ob.ice?.[lockTile]||keys.some(i=>ob.ice?.[i]||(design.down||[]).includes(i))||(design.down||[]).includes(lockTile))return {fail:'conflictingRoles'};
  if(design.faces[keys[0]]!==design.faces[keys[1]]||design.faces.filter(f=>f===design.faces[keys[0]]).length!==2)return {fail:'keysNotUniquePair'};
  const tiles=design.tiles.map(t=>Array.isArray(t)?{x:t[0],y:t[1],z:t[2]}:t);
  const visibility=require('../sim/refit/human_board_visibility.js').tileVisibility(tiles,{alive:new Uint8Array(tiles.length).fill(1),down:new Uint8Array(tiles.length),openPeek:-1},design.clock).tiles;
  if(keys.some(i=>!visibility[i].role))return {fail:'keysNotVisible'};
  if(!visibility[lockTile].role)return {fail:'lockNotVisible'};
  const limits={nodes:400000,totalNodes:700000,ms:Infinity,finishNodes:200000,finishMs:Infinity,...options};
  const before=E.clockShortest(E.game(design),limits);if(!before||before.unsure)return {fail:'sourceMinimumUnproven'};
  const copy=JSON.parse(JSON.stringify(design));copy.clock.n=99;copy.obst={...copy.obst,locks:[lockTile],keys:keys.slice()};
  const way=E.clockShortest(E.game(copy),limits);if(!way||way.unsure)return {fail:'minimumUnproven'};
  if(way.k<=before.k)return {fail:'idleLock',minimum:way.k,originalMinimum:before.k};
  if(way.k+spare>19)return {fail:'clockTooDeep',minimum:way.k};
  const strict=E.clockShortest(E.game(copy,{strict:true}),limits);
  if(!strict||strict.unsure||strict.k!==way.k)return {fail:'strictMinimumUnproven'};
  copy.clock.n=way.k+spare;copy.solution=E.flatten(strict.path);
  if(!E.replay(copy,copy.solution,{strict:true}).ok)return {fail:'witnessUnproven'};
  return {design:copy,proof:{keys:keys.slice(),lockTile,minimum:way.k,originalMinimum:before.k,spare,
    preserved:['tiles','faces','down','gold','obst.ice','clock.i'],changed:['obst.keys','obst.locks','clock.n','solution'],
    scope:'meaningful, visible key prelude; remaining puzzle/quality/difficulty require verification'}};
}
function clockKeyIce(design,keys,ice,spare,options={}) {
  const b=clockKeys(design,keys,spare,options);if(b.fail)return b;
  const t=ice?.tile,h=ice?.h;
  if(!Number.isInteger(t)||t<0||t>=design.tiles.length||![1,2].includes(h))return {fail:'invalidIce'};
  if(t===design.clock.i||keys.includes(t)||design.obst?.ice?.[t]||(design.down||[]).includes(t))return {fail:'conflictingRoles'};
  const tiles=design.tiles.map(p=>Array.isArray(p)?{x:p[0],y:p[1],z:p[2]}:p);
  const visible=require('../sim/refit/human_board_visibility.js').tileVisibility(tiles,{alive:new Uint8Array(tiles.length).fill(1),down:new Uint8Array(tiles.length),openPeek:-1},design.clock).tiles;
  if(!visible[t].role)return {fail:'iceNotVisible'};
  const d=b.design;d.obst.ice={...(d.obst.ice||{}),[t]:h};d.clock.n=99;
  const limits={nodes:400000,totalNodes:700000,ms:Infinity,finishNodes:200000,finishMs:Infinity,...options};
  const way=E.clockShortest(E.game(d),limits);if(!way||way.unsure)return {fail:'minimumUnproven'};
  if(way.k<=b.proof.minimum)return {fail:'idleIce',minimum:way.k,keyMinimum:b.proof.minimum};
  if(way.k+spare>19)return {fail:'clockTooDeep',minimum:way.k};
  const strict=E.clockShortest(E.game(d,{strict:true}),limits);
  if(!strict||strict.unsure||strict.k!==way.k)return {fail:'strictMinimumUnproven'};
  d.clock.n=way.k+spare;d.solution=E.flatten(strict.path);
  if(!E.replay(d,d.solution,{strict:true}).ok)return {fail:'witnessUnproven'};
  return {design:d,proof:{...b.proof,ice:{tile:t,h},keyMinimum:b.proof.minimum,minimum:way.k,
    preserved:['tiles','faces','down','gold','clock.i'],changed:['obst.keys','obst.locks','obst.ice','clock.n','solution']}};
}
// Re-pair only the prelude. At the declared later state all changed tiles are
// gone, their removed-picture multiset is identical, and the clock is gone.
// Thus every remaining picture, tray face, obstacle and future move is intact.
function framePictures(design,swaps,protectedState,spare,options={}) {
  if(!design.clock)return {fail:'noClock'};
  if(![1,2,3].includes(spare))return {fail:'invalidSpare'};
  if(!protectedState?.alive||protectedState.alive.length!==design.tiles.length||protectedState.alive[design.clock.i])return {fail:'invalidContinuation'};
  if(!Array.isArray(swaps)||swaps.some(p=>!Array.isArray(p)||p.length!==2||p.some(i=>!Number.isInteger(i)||i<0||i>=design.tiles.length||protectedState.alive[i]||(design.obst?.keys||[]).includes(i))))return {fail:'frameOutsideContract'};
  const d=JSON.parse(JSON.stringify(design));for(const [a,b] of swaps)[d.faces[a],d.faces[b]]=[d.faces[b],d.faces[a]];
  const removed=x=>x.faces.filter((_,i)=>!protectedState.alive[i]).slice().sort((a,b)=>a-b).join(',');
  if(removed(d)!==removed(design))return {fail:'continuationPictureMismatch'};
  d.clock.n=99;
  const limits={nodes:400000,totalNodes:700000,ms:Infinity,finishNodes:200000,finishMs:Infinity,...options};
  const way=E.clockShortest(E.game(d),limits);if(!way||way.unsure)return {fail:'minimumUnproven'};
  if(way.k+spare>19)return {fail:'clockTooDeep',minimum:way.k};
  const strict=E.clockShortest(E.game(d,{strict:true}),limits);if(!strict||strict.unsure||strict.k!==way.k)return {fail:'strictMinimumUnproven'};
  d.clock.n=way.k+spare;d.solution=E.flatten(strict.path);if(!E.replay(d,d.solution,{strict:true}).ok)return {fail:'witnessUnproven'};
  return {design:d,proof:{minimum:way.k,spare,swaps,continuation:{clockRemoved:true,unchangedAlivePictures:true,unchangedRemovedPictureMultiset:true},
    preserved:['tiles','down','gold','obst','clock.i','pictures of all tiles alive at the protected state'],changed:['prelude pictures','clock.n','solution']}};
}
module.exports = { clockSpare, clockTile, clockKeys, clockKeyIce, framePictures };
